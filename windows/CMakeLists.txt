cmake_minimum_required(VERSION 3.20)
project(audio_recorder_windows LANGUAGES CXX)

if(NOT WIN32)
  message(FATAL_ERROR "Use a Windows target with MSVC or LLVM-MinGW.")
endif()

add_executable(audio-recorder windows_recorder.cpp)
target_compile_features(audio-recorder PRIVATE cxx_std_17)
target_compile_definitions(audio-recorder PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(MSVC)
  target_compile_options(audio-recorder PRIVATE /W4 /EHsc)
elseif(MINGW)
  target_compile_options(audio-recorder PRIVATE -Wall -Wextra)
  target_link_options(audio-recorder PRIVATE -municode -static)
else()
  message(FATAL_ERROR "Supported Windows compilers are MSVC and LLVM-MinGW.")
endif()
target_link_libraries(audio-recorder PRIVATE ole32 uuid mmdevapi propsys)
